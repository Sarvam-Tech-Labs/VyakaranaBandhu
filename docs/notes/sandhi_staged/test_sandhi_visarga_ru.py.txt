"""
The रु, the visarga and the य् — `families/visarga_ru.py`.

Every expectation is one of the commentaries' own worked forms, with the
commentary named in the docstring, or a *kim* counter-example (where the
conditions of a sūtra live), or an assertion on the STEPS where a plain result
could come out right by luck (a rule that does nothing looks exactly like a
rule that was refused). A test that cannot fail is worse than no test
(NORTH_STAR §3): where a result depends on a flag, the same form is also run
without it, and where it depends on a `consumes` or an `overrides`, the
declaration is taken away and the result must change.

**What is run.** The family's own rules with the two families whose sūtras
finish its junctions (ac_ekadesa for 6.1.87, 6.1.109; hal_assimilation for
8.4.40 and 8.2.39), and ac_yan_ayadi where a 6.1.78 य्/व् is the subject. The
other families are not loaded, so a result does not move when they are added;
where a test wants the whole rulebook it says so and asserts only this
family's steps.

Sūtra numbers are never typed from memory in an assertion about a sūtra's
*words*: those come from the corpus (`test_the_sutras_named_are_the_corpus_words`).
"""

from __future__ import annotations

import dataclasses
import unittest
from typing import Iterable, List, Sequence, Tuple

from src.astadhyayi import corpus
from src.astadhyayi.anga import kharavasanayoh, sasajuso_ruh

    BHA_KUR_CHUR, UBHAYATHA_THREE, VASU_FOUR, the_word_end)
from src.astadhyayi.sandhi import rulebook, sandhi, trace
from src.astadhyayi.sandhi.engine import Outcome, derive
from src.astadhyayi.sandhi.families import visarga_ru as fam
from src.astadhyayi.sandhi.parse import parse, to_iast
from src.astadhyayi.sandhi.segs import RU, View

from src.astadhyayi.visarjaniya import (
    DVIS_THREE, KRKAMI_SEVEN, NIRADI, PATI_SEVEN, the_visarga)

#: The family, and the two whose sūtras finish its junctions.
RULES = rulebook.rules_of("visarga_ru", "ac_ekadesa", "hal_assimilation")
#: … with 6.1.78, for the य्/व् that a diphthong makes.
RULES_YAN = rulebook.rules_of("visarga_ru", "ac_ekadesa", "hal_assimilation",
                              "ac_yan_ayadi")
#: This family alone: what its rules do and nothing that follows.
ALONE = rulebook.rules_of("visarga_ru")


def run(text, rules=RULES, **kw):
    return sandhi(text, rules=rules, **kw)


def first(text, rules=RULES, **kw) -> Outcome:
    return run(text, rules, **kw).outcomes[0]


def forms(text, rules=RULES, **kw) -> set:
    return set(run(text, rules, **kw).surfaces)


def taken(outcome: Outcome) -> List[str]:
    """The sūtras of a derivation, in order — an option declined is not a step."""
    return [s.sutra for s in outcome.steps if not s.declined]


def at_junction(outcome: Outcome) -> List[str]:
    """`taken`, without the steps that touch only the LAST word — its own final
    स् at the pause (8.2.66, 8.3.15), which is not the junction being tested."""
    last = len(outcome.start.words) - 1
    return [s.sutra for s in outcome.steps
            if not s.declined and s.words != (last,)]


def steps(text, rules=RULES, **kw) -> List[str]:
    return at_junction(first(text, rules, **kw))


def every_course(text, rules=RULES, **kw) -> List[Tuple[str, List[str]]]:
    """(surface, sūtras taken) for each derivation the options give."""
    return [(o.surface, at_junction(o)) for o in run(text, rules, **kw).outcomes]


def after_step(outcome: Outcome, sutra: str, occurrence: int = 0) -> str:
    """The form as it stood after the `occurrence`-th step of that sūtra."""
    found = [s for s in outcome.steps if s.sutra == sutra and not s.declined]
    return found[occurrence].after


def step_of(outcome: Outcome, sutra: str):
    return next(s for s in outcome.steps if s.sutra == sutra and not s.declined)


def displaced(outcome: Outcome, sutra: str) -> List[str]:
    """The sūtras that the taken step of `sutra` displaced."""
    return [a.sutra for a in step_of(outcome, sutra).against]


class _Base(unittest.TestCase):
    def assertForms(self, text, expected, rules=RULES, **kw):
        self.assertEqual(forms(text, rules, **kw), set(expected), text)

    def assertSteps(self, text, expected, rules=RULES, **kw):
        self.assertEqual(steps(text, rules, **kw), list(expected), text)


# ---------------------------------------------------------------------------
# 8.2.66  ससजुषो रुः
# ---------------------------------------------------------------------------


class Rutva(_Base):
    """ससजुषो रुः — a pada-final स् (and the ष् of *sajuṣ*) becomes रु."""

    def test_kashika_agnir_atra_vayur_atra(self):
        """Kāśikā on 8.2.66: सकारान्तस्य — अग्निरत्र। वायुरत्र। (The रु is a plain
        र् before a vowel after इ or उ: 6.1.113 wants a short अ.)"""
        self.assertForms("agnis atra", {"agniratra"})
        self.assertForms("vāyus atra", {"vāyuratra"})
        self.assertSteps("agnis atra", ["8.2.66"])

    def test_the_ru_is_a_marked_r_and_not_the_r_of_punar(self):
        """The रु is `RU`-marked so that 6.1.113–114 can tell it from the र् of
        *punar*, which they do not touch."""
        out = first("agnis atra")
        made = [s for s in out.final.segs if s.made_by == "8.2.66"]
        self.assertEqual([(s.s, RU in s.marks) for s in made], [("r", True)])
        plain = parse("punar atra").segs
        self.assertTrue(all(RU not in s.marks for s in plain))

    def test_kashika_sajus_names_the_final_sa_of_sajus(self):
        """Kāśikā on 8.2.66: सजुषः — सजूर्ऋतुभिः (सजुष् + ऋतुभिः): the ष् of the
        word *sajuṣ* becomes रु."""
        self.assertForms("sajuṣ ṛtubhiḥ", {"sajurṛtubhiḥ"})
        self.assertEqual(steps("sajuṣ ṛtubhiḥ"), ["8.2.66"])

    def test_balamanorama_sajusau_sajusah_keep_their_sa(self):
        """Bālamanoramā on 8.2.66: सजुषौ, सजुषः — ष् does not end a पद, so no रु."""
        out = first("sajuṣ~au")
        self.assertNotIn("8.2.66", taken(out))
        self.assertEqual(out.surface, "sajuṣau")

    def test_any_other_final_sa_is_not_a_ru(self):
        """A pada-final ष् of a word that is not *sajuṣ* is not named."""
        self.assertNotIn("8.2.66", taken(first("viṣ atra")))

    def test_a_final_sa_that_is_not_the_end_of_a_pada_stays(self):
        """`~` says the two pieces are one pada: the first is not pada-final."""
        self.assertNotIn("8.2.66", taken(first("agnis~a")))

    def test_it_displaces_jastva(self):
        """Kaumudī on 8.2.66: जश्त्वापवादः — शिवस् would otherwise have its स् made
        द् by 8.2.39 (Bālamanoramā: ननु शिवस् इति सकारस्य झलां जशोऽन्ते …)."""
        out = first("śivas ca")
        self.assertEqual(taken(out)[0], "8.2.66")
        self.assertNotIn("8.2.39", taken(out))
        self.assertIn("8.2.39", displaced(out, "8.2.66"))
        why = next(a.why for a in step_of(out, "8.2.66").against
                   if a.sutra == "8.2.39")
        self.assertIn("जश्त्वापवादः", why)

    def test_without_the_declaration_jastva_takes_the_place(self):
        """The override is what does it: take it away and 8.2.39, which stands
        before 8.2.66 in the tripādī, acts first and makes the स् a द् (the
        very fault Bālamanoramā names) — the रु never comes."""
        stripped = tuple(dataclasses.replace(r, overrides=())
                         if r.sutra == "8.2.66" else r for r in RULES)
        out = first("śivas ca", stripped)
        self.assertEqual(taken(out)[0], "8.2.39")
        self.assertNotEqual(out.surface, first("śivas ca").surface)

    def test_agrees_with_the_projects_own_8_2_66(self):
        """`anga.sasajuso_ruh` is 8.2.66 as a question; the engine's step is the
        same sūtra as a derivation. Where the first says a word takes रु the
        engine's first step is 8.2.66, and where it says not, the engine has no
        such step."""
        for word in ("rāmas", "harīs", "agnis", "manas", "viśvatas"):
            self.assertIsNotNone(sasajuso_ruh(word).result, word)
            self.assertEqual(steps(f"{word} ca")[0], "8.2.66", word)
        for word in ("rāma", "hariḥ", "punar"):
            if not word.endswith("s"):
                self.assertIsNone(sasajuso_ruh(word).result, word)
        self.assertNotIn("8.2.66", steps("rāma ca"))
        self.assertNotIn("8.2.66", steps("punar ca"))

# 6.1.113–114  the रु as उ
# ---------------------------------------------------------------------------


class RuAsU(_Base):
    """अतो रोरप्लुतादप्लुते, हशि च — the रु, after a short अ, is उ."""

    def test_kashika_vrksa_atra_plaksa_atra(self):
        """Kāśikā on 6.1.113: वृक्षोऽत्र। प्लक्षोऽत्र। — with the steps: the रु
        (8.2.66), the उ (6.1.113), ओ (6.1.87), and the avagraha (6.1.109)."""
        self.assertForms("vṛkṣas atra", {"vṛkṣo'tra"})
        self.assertForms("plakṣas atra", {"plakṣo'tra"})
        self.assertSteps("vṛkṣas atra",
                         ["8.2.66", "6.1.113", "6.1.87", "6.1.109"])

    def test_laghu_sivo_rcyah(self):
        """Laghusiddhāntakaumudī on 6.1.113: शिवोऽर्च्यः."""
        self.assertForms("śivas arcyaḥ", {"śivo'rcyaḥ"})
        self.assertEqual(steps("śivas arcyaḥ"),
                         ["8.2.66", "6.1.113", "6.1.87", "6.1.109"])

    def test_it_displaces_the_ya_of_8_3_17(self):
        """Kaumudī on 6.1.113: भोभगोअघो इति प्राप्तस्य यत्वस्यापवादः — the step
        says so, and without the declaration 8.3.17's य् is not shown displaced."""
        out = first("vṛkṣas atra")
        self.assertIn("8.3.17", displaced(out, "6.1.113"))
        self.assertNotIn("8.3.17", taken(out))
        stripped = tuple(dataclasses.replace(r, overrides=())
                         if r.sutra == "6.1.113" else r for r in RULES)
        self.assertNotIn("8.3.17", displaced(first("vṛkṣas atra", stripped),
                                             "6.1.113"))

    def test_kashika_agnir_atra_has_no_short_a_before(self):
        """Kāśikā: अत इति किम्? अग्निरत्र — इ before the रु."""
        self.assertNotIn("6.1.113", taken(first("agnis atra")))
        self.assertForms("agnis atra", {"agniratra"})

    def test_kashika_vrksa_atra_the_tapara_of_a(self):
        """Kāśikā: तपरकरणं किम्? वृक्षा अत्र — a long आ before the रु: 8.3.17's
        य्, not the उ (and then 8.3.19's option)."""
        out = first("vṛkṣās atra")
        self.assertNotIn("6.1.113", taken(out))
        self.assertIn("8.3.17", taken(out))
        self.assertForms("vṛkṣās atra", {"vṛkṣāatra", "vṛkṣāyatra"})

    def test_kashika_svar_atra_pratar_atra_are_not_a_ru(self):
        """Kāśikā: सानुबन्धग्रहणं किम्? स्वरत्र। प्रातरत्र। — a plain र् is not the
        रु of 8.2.66, whatever stands beside it."""
        for text in ("svar atra", "prātar atra"):
            self.assertNotIn("6.1.113", taken(first(text)), text)
            self.assertEqual(forms(text), {text.replace(" ", "")}, text)

    def test_kashika_vrksa_iha_no_short_a_after(self):
        """Kāśikā: अतीत्येव — वृक्ष इह: the sound after is इ, not अ. (The रु is
        8.3.17's य्, which 8.3.19 may drop.)"""
        out = first("vṛkṣas iha")
        self.assertNotIn("6.1.113", taken(out))
        self.assertEqual(forms("vṛkṣas iha"), {"vṛkṣaiha", "vṛkṣayiha"})

    def test_kashika_vrksa_asritah_the_tapara_of_the_following_a(self):
        """Kāśikā: तस्यापि तपरत्वादत्र न भवति — वृक्ष आश्रितः (a long आ follows)."""
        self.assertNotIn("6.1.113", taken(first("vṛkṣas āśritaḥ")))

    def test_kashika_pluta_before_and_after(self):
        """Kāśikā: अप्लुतादिति किम्? सुस्रोत३ अत्र न्वसि; अप्लुत इति किम्? तिष्ठतु पय
        अ३श्विन्. The letters cannot say a vowel is pluta: the flag `pluta`,
        or the mark a rule of 8.2.82–107 leaves."""
        self.assertIn("6.1.113", taken(first("susrotas atra")))
        self.assertNotIn("6.1.113", taken(first("susrotas{pluta} atra")))
        # the same after: `a` flagged pluta
        self.assertIn("6.1.113", taken(first("payas a")))
        self.assertNotIn("6.1.113", taken(first("payas a{pluta}")))

    def test_the_pluta_mark_is_also_read(self):
        """A vowel a rule has *made* pluta carries `PLUTA` — 6.1.113 reads it."""
        from src.astadhyayi.sandhi.segs import PLUTA
        state = parse("payas atra")
        segs = list(state.segs)
        pre = [i for i, s in enumerate(segs) if s.w == 0][-2]   # अ before स्
        self.assertEqual(segs[pre].s, "a")
        marked = list(segs)
        marked[pre] = dataclasses.replace(segs[pre],
                                          marks=segs[pre].marks | {PLUTA})
        self.assertIn("6.1.113", taken(derive(state, RULES)[0]))
        out = derive(dataclasses.replace(state, segs=tuple(marked)), RULES)[0]
        self.assertNotIn("6.1.113", taken(out))

    def test_kashika_purusa_yati_hasati_dadati(self):
        """Kāśikā on 6.1.114: पुरुषो याति। पुरुषो हसति। पुरुषो ददाति।"""
        for word, expected in (("yāti", "puruṣoyāti"), ("hasati", "puruṣohasati"),
                               ("dadāti", "puruṣodadāti")):
            self.assertForms(f"puruṣas {word}", {expected})
            self.assertSteps(f"puruṣas {word}",
                             ["8.2.66", "6.1.114", "6.1.87"])

    def test_laghu_sivo_vandyah(self):
        """Laghusiddhāntakaumudī on 6.1.114: शिवो वन्द्यः."""
        self.assertForms("śivas vandyaḥ", {"śivovandyaḥ"})

    def test_kaumudi_pratar_atra_bhratar_gaccha(self):
        """Kaumudī on 6.1.114: रोरित्युकारानुबन्धग्रहणान्नेह — प्रातरत्र, भ्रातर्गच्छ:
        a plain र् before अ or before a हश् keeps its र्."""
        self.assertForms("bhrātar gaccha", {"bhrātargaccha"})
        self.assertNotIn("6.1.114", taken(first("bhrātar gaccha")))

    def test_kaumudi_devas_iha_the_long_a(self):
        """Kaumudī on 6.1.114: देवास् इह इति स्थिते। रुत्वम् — आ is not a short
        अ, so not the उ."""
        self.assertNotIn("6.1.114", taken(first("devās iha")))

    def test_the_ru_is_seen_only_because_the_rule_names_it(self):
        """8.2.66 stands in the tripādī, so 8.2.1 hides its रु from a sapāda
        rule; 6.1.113 and 6.1.114 name the रु and so `consumes` it. Take that
        away and neither can act (the engine's own test does the same for
        6.1.113; here both)."""
        for sutra, text in (("6.1.113", "vṛkṣas atra"),
                            ("6.1.114", "puruṣas dadāti")):
            blind = tuple(dataclasses.replace(r, consumes=frozenset())
                          if r.sutra == sutra else r for r in RULES)
            out = first(text, blind)
            self.assertNotIn(sutra, taken(out), sutra)
            self.assertNotEqual(out.surface, first(text).surface, sutra)

    def test_manoratha_6_1_114_against_8_3_14(self):
        """Laghusiddhāntakaumudī on 1.4.2: पूर्वत्रासिद्धमिति रोरीत्यसिद्धत्वादुत्वमेव
        — मनोरथः. Both reach the रु of *manas ratha*; 8.3.14 is later in the
        tripādī, so 8.2.1 gives the उ to 6.1.114 and 8.3.14 is not taken."""
        self.assertForms("manas ratha", {"manoratha"})
        self.assertSteps("manas ratha", ["8.2.66", "6.1.114", "6.1.87"])
        self.assertNotIn("8.3.14", taken(first("manas ratha")))

    def test_the_two_are_the_same_place_for_different_sounds(self):
        """6.1.113 is for a short अ after, 6.1.114 for a हश्: an अ is no हश्."""
        self.assertIn("6.1.113", taken(first("vṛkṣas atra")))
        self.assertNotIn("6.1.114", taken(first("vṛkṣas atra")))
        self.assertIn("6.1.114", taken(first("vṛkṣas gacchati")))
        self.assertNotIn("6.1.113", taken(first("vṛkṣas gacchati")))

class SuLopa(_Base):
    """एतत्तदोः सुलोपोऽकोरनञ्समासे हलि; स्यश्छन्दसि बहुलम्; सोऽचि लोपे चेत्
    पादपूरणम् — and the heading 6.1.135."""

    def test_laghu_esa_visnuh_sa_sambhuh(self):
        """Laghusiddhāntakaumudī on 6.1.132: एष विष्णुः। स शंभुः। (Kāśikā: एष ददाति,
        स ददाति, एष भुङ्क्ते, स भुङ्क्ते.) The सु is lost before a consonant, and
        8.2.66 then finds no स् to make a रु of."""
        for text, form in (("eṣas viṣṇuḥ", "eṣaviṣṇuḥ"),
                           ("sas śambhuḥ", "saśambhuḥ"),
                           ("eṣas dadāti", "eṣadadāti"),
                           ("sas dadāti", "sadadāti"),
                           ("eṣas bhuṅkte", "eṣabhuṅkte"),
                           ("sas bhuṅkte", "sabhuṅkte")):
            self.assertForms(text, {form})
            self.assertEqual(steps(text), ["6.1.132"], text)

    def test_the_su_is_lost_before_the_ru_is_made(self):
        """6.1.132 is sapādasaptādhyāyī, 8.2.66 tripādī: the loss acts first and
        8.2.66 acts on what it leaves — nothing. So the trace has no रु."""
        out = first("eṣas viṣṇuḥ")
        self.assertNotIn("8.2.66", [s.sutra for s in out.steps
                                     if s.words != (1,)])

    def test_laghu_hali_kim_eso_tra(self):
        """Laghusiddhāntakaumudī: हलि किम्? एषोऽत्र — a vowel follows, so the स्
        goes on to रु (8.2.66), उ (6.1.113), ओ (6.1.87), and the avagraha."""
        self.assertForms("eṣas atra", {"eṣo'tra"})
        self.assertForms("sas atra", {"so'tra"})
        self.assertEqual(steps("eṣas atra"),
                         ["8.2.66", "6.1.113", "6.1.87", "6.1.109"])

    def test_laghu_akoh_kim_esako_rudrah(self):
        """Laghusiddhāntakaumudī: अकोः किम्? एषको रुद्रः — the सकच् form: the सु
        stays and goes to रु and उ (6.1.114). The flag `akac` says the word has
        the affix क, which the letters do not; without it the सु WOULD go."""
        self.assertForms("eṣakas rudraḥ", {"eṣakorudraḥ"})
        self.assertNotIn("6.1.132", steps("eṣakas{stem:etad,akac} rudraḥ"))
        self.assertForms("eṣakas{stem:etad,akac} rudraḥ", {"eṣakorudraḥ"})
        # …the same form without `akac` is the ordinary word, and loses it:
        self.assertIn("6.1.132", steps("eṣakas{stem:etad} rudraḥ"))

    def test_laghu_ananasamase_kim_asah_sivah(self):
        """Laghusiddhāntakaumudī: अनञ्समासे किम्? असः शिवः — the negative
        compound keeps its सु (flag `nan_samasa`); it becomes रु and visarga."""
        self.assertNotIn("6.1.132", steps("asas{stem:tad,nan_samasa} śivaḥ"))
        self.assertIn("6.1.132", steps("asas{stem:tad} śivaḥ"))
        self.assertForms("asas{stem:tad,nan_samasa} śivaḥ", {"asaśśivaḥ",
                                                             "asaḥśivaḥ"})

    def test_kashika_etattador_iti_kim_yo_dadati(self):
        """Kāśikā: एतत्तदोरिति किम्? यो ददाति — a word that is not एतद्/तद् takes
        रु and उ (6.1.114) instead."""
        self.assertNotIn("6.1.132", steps("yas dadāti"))
        self.assertForms("yas dadāti", {"yodadāti"})

    def test_stem_flag_names_another_form_of_etad(self):
        """The two nominatives *eṣas*, *sas* are recognised by their letters;
        any other form is the caller's `stem:etad` / `stem:tad`."""
        self.assertIn("6.1.132", steps("tas{stem:tad} dadāti"))
        self.assertNotIn("6.1.132", steps("tas dadāti"))

    def test_the_su_goes_only_before_a_consonant(self):
        """No loss before a vowel or at a pause (संहितायाम् — a pause is not a
        junction): एषः, सः."""
        self.assertNotIn("6.1.132", taken(first("eṣas")))

    def test_laghu_anah_kim_trdhah_vrdhah(self):

        for text, form in (("tṛḍh~ḍha", "tṛḍha"), ("vṛḍh~ḍha", "vṛḍha"),
                           ("ātṛḍh~ḍha", "ātṛḍha")):
            self.assertForms(text, {form})
            self.assertEqual(steps(text), ["8.3.13"], text)

    def test_kashika_svalid_dhaukate_has_no_dho_dhe(self):
        """Kāśikā on 8.3.13: श्वलिड् ढौकते — the first ढ् is a ड् by 8.2.39 and
        ends a pada, so there is nothing for the rule to take (अपदान्तस्य)."""
        out = first("śvaliḍ ḍhaukate")
        self.assertNotIn("8.3.13", taken(out))
        self.assertEqual(out.surface, "śvaliḍḍhaukate")

    def test_a_pada_final_dha_is_not_lost(self):
        """The Kāśikā reads the sūtra of a ढ् that does NOT end a pada."""
        self.assertNotIn("8.3.13", taken(first("liḍh ḍhaukate")))

    def test_kashika_ro_ri_in_a_pada_not_only_at_its_end(self):
        """Kāśikā on 8.3.14: पदस्येत्यत्र विशेषणे षष्ठी, तेनापदान्तस्यापि रेफस्य
        लोपो भवति — जर्गृधेरजर्घाः: the र् of the root, not at the end of the
        pada, is lost before the रु that 8.2.75 has made (here the two pieces
        of one pada, `~`), and the अ before it is long (6.3.111)."""
        out = first("ajarghar~r")
        self.assertEqual(taken(out)[:2], ["8.3.14", "6.3.111"])
        self.assertEqual(out.surface, "ajarghāḥ")
        # and the र् of a preverb — निस् + रक्तम्: नीरक्तम्

        self.assertIn("8.3.14", taken(out))
        self.assertEqual(out.surface, "nīraktam")

    def test_ro_ri_is_only_before_a_ra(self):
        """A र् before anything else is not lost: अग्निर् नयति."""
        self.assertNotIn("8.3.14", taken(first("punar nayati")))

    def test_the_vowel_is_lengthened_by_the_projects_own_dirgha(self):
        """6.3.111's length is the one `anga.akah_savarne_dirghah` gives an अण्

            step = step_of(out, "6.3.111")
            self.assertEqual(step.detail.adesa, vowel, text)
            self.assertIn("1.1.66", step.sutras)

class VisargaOfRu(_Base):
    """खरवसानयोर्विसर्जनीयः, रोः सुपि."""

    def test_kashika_vrksa_chadayati_plaksa_chadayati(self):
        """Kāśikā on 8.3.15: वृक्षश्छादयति। प्लक्षश्छादयति। वृक्षस्तरति। प्लक्षस्तरति —
        the visarga (8.3.15), then स् (8.3.34), and 8.4.40 before a च-वर्ग."""
        self.assertForms("vṛkṣas chādayati", {"vṛkṣaśchādayati"})
        self.assertForms("plakṣas chādayati", {"plakṣaśchādayati"})
        self.assertForms("vṛkṣas tarati", {"vṛkṣastarati"})
        self.assertForms("plakṣas tarati", {"plakṣastarati"})
        self.assertSteps("vṛkṣas tarati", ["8.2.66", "8.3.15", "8.3.34"])
        self.assertSteps("vṛkṣas chādayati",
                         ["8.2.66", "8.3.15", "8.3.34", "8.4.40"])

    def test_laghu_ramas_ca(self):
        """Laghusiddhāntakaumudī (and the engine's own test): रामस् च — the four
        steps 8.2.66, 8.3.15, 8.3.34, 8.4.40, and रामश्च."""
        self.assertForms("rāmas ca", {"rāmaśca"})
        self.assertSteps("rāmas ca", ["8.2.66", "8.3.15", "8.3.34", "8.4.40"])

    def test_kashika_avasane_vrksah_plaksah(self):
        """Kāśikā on 8.3.15: अवसाने — वृक्षः। प्लक्षः। — at a pause; the visarga
        of a pause is no जिह्वामूलीय and no स् (the Bhāṣya on 8.3.34: संहितायाम्)."""
        self.assertForms("vṛkṣas", {"vṛkṣaḥ"})
        self.assertForms("plakṣas", {"plakṣaḥ"})
        self.assertEqual(taken(first("vṛkṣas")), ["8.2.66", "8.3.15"])
        step = step_of(first("vṛkṣas"), "8.3.15")
        self.assertIn("1.4.110", step.sutras)     # the pause is a condition

    def test_the_visarga_of_a_pause_is_not_a_junction_of_words(self):
        """No pause after the last word (`pause=False`, the utterance goes on):
        no visarga."""
        out = first("vṛkṣas", pause=False)
        self.assertNotIn("8.3.15", taken(out))

    def test_kashika_kharavasanayoriti_kim_agnir_nayati(self):
        """Kāśikā on 8.3.15: खरवसानयोरिति किम्? अग्निर्नयति। वायुर्नयति — न् is neither
        a खर् nor a pause."""
        self.assertForms("agnis nayati", {"agnirnayati"})
        self.assertForms("vāyus nayati", {"vāyurnayati"})
        self.assertNotIn("8.3.15", taken(first("agnis nayati")))

    def test_a_plain_r_is_a_visarga_too(self):
        """The र् of *punar* (not a रु) is also made visarga before a खर्: पुनः
        तत्र — and the र् before a vowel stays."""
        self.assertIn("8.3.15", steps("punar tatra"))
        self.assertNotIn("8.3.15", steps("punar atra"))

    def test_the_r_of_a_vowel_join_is_not_pada_final(self):
        """Engine README (6.1.85 is not applied to a rule of sounds): the र् of
        अर् in *महर्षि* is not a pada-final र्, so no visarga."""
        self.assertNotIn("8.3.15", taken(first("mahā ṛṣi")))

    def test_kashika_payahsu_yasahsu_niyama(self):
        """Kāśikā on 8.3.16: पयःसु। यशःसु — the रु before the locative plural सु
        is a visarga; and the स् of सु is what 8.3.34 or 8.3.36 then work on."""
        for text in ("payas su{saptamibahuvacana}", "yaśas su{saptamibahuvacana}"):
            self.assertIn("8.3.15", steps(text), text)
        self.assertEqual(forms("payas su{saptamibahuvacana}"),
                         {"payaḥsu", "payassu"})

    def test_kashika_girsu_dhursu_the_niyama_refuses(self):
        """Kāśikā on 8.3.16: रोरेव सुपि विसर्जनीयादेशः, नान्यस्य। गीर्षु। धूर्षु — a
        plain र् before सु is not a visarga."""
        for text in ("gīr su{saptamibahuvacana}", "dhūr su{saptamibahuvacana}"):
            out = first(text)
            self.assertNotIn("8.3.15", taken(out), text)
            self.assertIn("8.3.16", taken(out), text)
            self.assertEqual(out.surface, text.split()[0] + "su", text)
        step = step_of(first("gīr su{saptamibahuvacana}"), "8.3.16")
        self.assertEqual(step.detail.kind, "pratiṣedha")
        self.assertEqual(step.before, step.after)
        self.assertIn("8.3.15", [a.sutra for a in step.against])

    def test_the_refusal_is_at_the_very_site_of_8_3_15(self):
        """README: a refusal names the SAME site as the rule it refuses."""
        state = parse("gīr su{saptamibahuvacana}")
        by = {r.sutra: r for r in RULES}
        view = lambda r: View(state, r.sutra, r.consumes)
        refusal = list(by["8.3.16"].find(view(by["8.3.16"])))
        khar = list(by["8.3.15"].find(view(by["8.3.15"])))
        self.assertEqual(len(refusal), 1)
        self.assertEqual(len(khar), 1)
        self.assertEqual(refusal[0].site, khar[0].site)
        self.assertEqual(refusal[0].edits, ())

    def test_without_the_su_flag_the_niyama_is_silent(self):
        """Without `saptamibahuvacana` the letters s-u are not known to be the
        ending, and 8.3.15 acts as it does before any खर्: गीर् सु."""
        out = first("gīr su")
        self.assertIn("8.3.15", taken(out))
        self.assertNotIn("8.3.16", taken(out))

    def test_the_ru_before_su_cites_the_niyama(self):
        """The visarga of a रु before सु is 8.3.15's, and its step says the
        niyama of 8.3.16 allows exactly that."""
        step = step_of(first("payas su{saptamibahuvacana}"), "8.3.15")
        self.assertIn("8.3.16", step.sutras)
        plain = step_of(first("punar tatra"), "8.3.15")
        self.assertNotIn("8.3.16", plain.sutras)

    def test_agrees_with_the_projects_own_8_3_15(self):
        """`anga.kharavasanayoh` is 8.3.15 as a question. Where it says a word's
        र् becomes the visarga before a खर् or a pause, the engine's derivation
        has that step; where it says not, it has not."""
        cases = (("vṛkṣar", "tarati", True, True), ("vṛkṣar", "", True, False),
                 ("agnir", "nayati", False, True), ("punar", "atra", False, True))
        for word, following, pause_ok, has_next in cases:
            asked = kharavasanayoh(word, at_pause=not has_next,
                                   following=following)
            text = f"{word} {following}".strip()
            asserted = "8.3.15" in taken(first(text.replace("vṛkṣar", "vṛkṣas")
                                               if word == "vṛkṣar" else text))
            self.assertEqual(asserted, asked.result is not None, text)

class YaAndItsFates(_Base):
    """भोभगोअघोअपूर्वस्य योऽशि; व्योर्लघुप्रयत्नतरः शाकटायनस्य; लोपः शाकल्यस्य; ओतो

    def test_laghu_deva_iha_devayiha(self):

        before an अश् is य् (8.3.17), and Śākalya's option drops it (8.3.19)."""
        self.assertForms("devās iha", {"devāiha", "devāyiha"})
        courses = dict(every_course("devās iha"))
        self.assertEqual(courses["devāiha"], ["8.2.66", "8.3.17", "8.3.19"])
        self.assertEqual(courses["devāyiha"], ["8.2.66", "8.3.17"])

    def test_the_loss_of_the_ya_does_not_let_the_vowels_join(self):
        """8.2.1: the loss is asiddha to the sandhi of vowels — देवा इह keeps its
        two vowels apart (Kaumudī on 8.3.19: पूर्वत्रासिद्धमिति लोपशास्त्रस्यासिद्धत्वान्न
        स्वरसन्धिः)."""
        out = next(o for o in run("devās iha").outcomes
                   if "8.3.19" in taken(o))
        self.assertEqual(out.text(), "devā iha")
        self.assertNotIn("6.1.101", taken(out))

    def test_laghu_bhos_bhagos_aghos(self):
        """Laghusiddhāntakaumudī on 8.3.22: भो देवाः। भगो नमस्ते। अघो याहि — and
        Kaumudī on 8.3.17: भोस् भगोस् अघोस् इति सकारान्ता निपाताः, तेषां रोर्यत्वे
        कृते. The य् before a consonant is lost by all (8.3.22)."""
        for text, form in (("bhos devāḥ", "bhodevāḥ"),
                           ("bhagos namaste", "bhagonamaste"),
                           ("aghos yāhi", "aghoyāhi"),
                           ("bhos lakṣmi", "bholakṣmi"),
                           ("bhos vidvadvṛnda", "bhovidvadvṛnda")):
            self.assertForms(text, {form})
            self.assertEqual(steps(text), ["8.2.66", "8.3.17", "8.3.22"], text)

    def test_kashika_bho_atra_bhago_atra_agho_atra(self):
        """Kāśikā on 8.3.17: भो अत्र। भगो अत्र। अघो अत्र। — before a vowel the य् is
        lost, always, by Gārgya (8.3.20)."""
        for text, form in (("bhos atra", "bhoatra"), ("bhagos atra", "bhagoatra"),
                           ("aghos atra", "aghoatra"), ("bhos idam", "bhoidam")):
            self.assertForms(text, {form})
            self.assertEqual(steps(text), ["8.2.66", "8.3.17", "8.3.20"], text)
            self.assertEqual(len(run(text).outcomes), 1, text)   # no option

    def test_the_o_of_bho_is_not_joined_to_the_vowel_after(self):
        """Kaumudī on 8.3.17: असन्धिः सौत्रः — भो अत्र stays apart (6.1.109 would
        give *भोऽत्र*; the loss is asiddha to it)."""
        out = first("bhos atra")
        self.assertEqual(out.text(), "bho atra")
        self.assertNotIn("6.1.109", taken(out))

    def test_kashika_brahmana_dadati_purusa_dadati(self):
        """Kāśikā on 8.3.17: ब्राह्मणा ददति। पुरुषा ददति — an अवर्ण before the रु
        and a consonant अश् (द्) after: the य् is lost (8.3.22)."""
        self.assertForms("brāhmaṇās dadati", {"brāhmaṇādadati"})
        self.assertForms("puruṣās dadati", {"puruṣādadati"})
        self.assertEqual(steps("brāhmaṇās dadati"),
                         ["8.2.66", "8.3.17", "8.3.22"])
        self.assertEqual(len(run("brāhmaṇās dadati").outcomes), 1)

    def test_kashika_ka_aste_kayaste(self):
        """Kāśikā on 8.3.17 and 8.3.19: क आस्ते, कयास्ते; काक आस्ते, काकयास्ते
        — the अवर्ण-पूर्व रु's य् is dropped or kept."""
        self.assertForms("kas āste", {"kaāste", "kayāste"})
        self.assertForms("kākas āste", {"kākaāste", "kākayāste"})

    def test_kashika_agnir_atra_vayur_atra_the_other_sounds_before(self):
        """Kāśikā: भोभगोअघोअपूर्वस्येति किम्? अग्निरत्र। वायुरत्र — इ, उ before."""
        self.assertNotIn("8.3.17", taken(first("agnis atra")))
        self.assertNotIn("8.3.17", taken(first("vāyus atra")))

    def test_kashika_asi_kim_vrksah_and_kaumudi_devah_santi(self):
        """Kāśikā: अश्ग्रहणं किम्? वृक्षः। प्लक्षः; Kaumudī: अशि किम्? देवाः सन्ति —

        out = first("devās santi")

    def test_the_laghu_ya_is_offered_on_request(self):
        """Kāśikā on 8.3.18: भोयत्र, भो अत्र; कयास्ते, क आस्ते — Śākaṭāyana's
        lighter य्. It is spelt as य्, so it is offered only where the caller
        asks (`sakatayana`): three courses — the lighter sound, the loss,
        neither (Nyāsa on 8.3.19: त्रीणि रूपाणि)."""
        outs = run("kas{sakatayana} āste").outcomes
        self.assertEqual(len(outs), 3)
        lighter = [o for o in outs if "8.3.18" in taken(o)]
        self.assertEqual(len(lighter), 1)
        marked = [s for s in lighter[0].final.segs if fam.LAGHU in s.marks]
        self.assertEqual([s.s for s in marked], ["y"])
        self.assertEqual(sorted(o.surface for o in outs),
                         sorted(["kaāste", "kayāste", "kayāste"]))
        # without the flag there is no lighter course and two derivations
        self.assertEqual(len(run("kas āste").outcomes), 2)
        self.assertTrue(all("8.3.18" not in taken(o)
                            for o in run("kas āste").outcomes))

    def test_the_laghu_ya_stays_after_o_and_the_plain_one_is_lost(self):
        """Kāśikā on 8.3.20: नित्यार्थोऽयमारम्भः ... लघुप्रयत्नतरस्तु भवत्येव यकारः, भो
        अत्र, भोयत्र — after ओ the plain य् is always lost, the lighter one stays."""
        outs = run("bhos{sakatayana} atra").outcomes
        self.assertEqual(sorted(o.surface for o in outs), ["bhoatra", "bhoyatra"])
        stays = next(o for o in outs if o.surface == "bhoyatra")
        self.assertIn("8.3.18", taken(stays))
        self.assertNotIn("8.3.20", taken(stays))
        lost = next(o for o in outs if o.surface == "bhoatra")
        self.assertIn("8.3.20", taken(lost))

    def test_kashika_8_3_22_sarvesam_takes_the_laghu_ya_too(self):
        """Kāśikā on 8.3.22: सर्वेषांग्रहणं शाकटायनस्यापि लोपो यथा स्यात्, लघुप्रयत्नतरो
        मा भूदिति — before a consonant even Śākaṭāyana's य् is lost: one course."""

        self.assertEqual({o.surface for o in outs}, {"bhodevāḥ"})
        self.assertEqual(len(outs), 1)
        self.assertIn("8.3.18", [a.sutra for a in
                                 step_of(outs[0], "8.3.22").against])

    def test_8_3_19_does_not_lose_the_lighter_sound(self):
        """The lighter य् is the other course of the option, not a sound the loss
        then takes (Nyāsa: three courses, not four): with `sakatayana` no course
        both makes it lighter and drops it."""
        for o in run("kas{sakatayana} āste").outcomes:
            self.assertFalse({"8.3.18", "8.3.19"} <= set(taken(o)))

    def test_the_y_v_of_8_3_18_22_are_read_from_the_sutra(self):

        letter of *ओतः* too — not typed."""
        self.assertEqual(fam._yv(), ("v", "y"))
        self.assertEqual(fam._o(), "o")

class VisargaBeforeKhar(_Base):
    """विसर्जनीयस्य सः, शर्परे विसर्जनीयः, वा शरि, कुप्वोः ≍क≍पौ च."""

    def test_kashika_8_3_34_the_sa_before_each_khar(self):
        """Kāśikā on 8.3.34: वृक्षश्छादयति। प्लक्षश्छादयति। वृक्षष्ठकारः। प्लक्षष्ठकारः।
        वृक्षस्थकारः। प्लक्षस्थकारः। वृक्षश्चिनोति। प्लक्षश्चिनोति। वृक्षष्टीकते।
        प्लक्षष्टीकते। वृक्षस्तरति। प्लक्षस्तरति — the स् of 8.3.34 is what
        8.4.40 and 8.4.41 then turn to श् or ष् (8.4.41 is the hal family's:
        here it is the स् that is asserted, by the step)."""
        for text in ("vṛkṣas chādayati", "vṛkṣas ṭhakāraḥ", "vṛkṣas thakāraḥ",
                     "vṛkṣas cinoti", "vṛkṣas ṭīkate", "vṛkṣas tarati",
                     "plakṣas chādayati", "plakṣas ṭhakāraḥ"):
            out = first(text)
            self.assertIn("8.3.34", taken(out), text)
            self.assertTrue(after_step(out, "8.3.34").startswith(
                text.split()[0][:-1] + "s "), text)

    def test_laghu_visnustrata(self):
        """Laghusiddhāntakaumudī on 8.3.34: खरि — विष्णुस्त्राता."""
        self.assertForms("viṣṇus trātā", {"viṣṇustrātā"})
        self.assertSteps("viṣṇus trātā", ["8.2.66", "8.3.15", "8.3.34"])

    def test_bhashya_a_pause_is_no_khar(self):
        """Bhāṣya on 8.3.34: इह कस्मान्न भवति — वृक्षः प्लक्ष इति? — the visarga of
        a pause stays (संहितायाम्; nothing follows)."""
        self.assertNotIn("8.3.34", taken(first("vṛkṣas")))
        self.assertForms("vṛkṣas", {"vṛkṣaḥ"})

    def test_the_utsarga_does_not_return_in_the_field_of_8_3_37(self):
        """Kaumudī on 8.3.37: येन नाप्राप्त इति न्यायेन विसर्जनीयस्य सः इत्यस्यापवादः.
        Declining 8.3.37's option leaves the visarga (चाद्विसर्गः), never the स्:

        got = forms("vṛkṣas karoti")
        self.assertEqual(got, {"vṛkṣaẖkaroti", "vṛkṣaḥkaroti"})
        self.assertNotIn("vṛkṣaskaroti", got)
        for p in ("vṛkṣas pacati",):
            self.assertNotIn("vṛkṣaspacati", forms(p))

    def test_kashika_8_3_37_vrksa_karoti_khanati_pacati_phalati(self):
        """Kāśikā on 8.3.37: वृक्ष≍ करोति, वृक्षः करोति। वृक्ष≍ खनति, वृक्षः खनति।
        वृक्ष≍ पचति, वृक्षः पचति। वृक्ष≍ फलति, वृक्षः फलति — the कवर्ग takes the
        जिह्वामूलीय, the पवर्ग the उपध्मानीय (1.3.10), or the visarge stays."""
        for text, special in (("vṛkṣas karoti", JIHVAMULIYA),
                              ("vṛkṣas khanati", JIHVAMULIYA),
                              ("vṛkṣas pacati", UPADHMANIYA),
                              ("vṛkṣas phalati", UPADHMANIYA)):
            stem, word = text.split()
            self.assertEqual(forms(text), {f"vṛkṣa{special}{word}",
                                           f"vṛkṣa{VISARGA}{word}"}, text)

    def test_the_two_substitutes_are_paired_by_1_3_10(self):
        """कुप्वोः ≍क≍पौ — two lists of equal length, so 1.3.10 pairs them in

        step = step_of(first("vṛkṣas karoti"), "8.3.37")
        self.assertIn("1.3.10", step.sutras)
        self.assertEqual(step.detail.adesa, JIHVAMULIYA)
        self.assertEqual(step_of(first("vṛkṣas pacati"), "8.3.37")
                         .detail.adesa, UPADHMANIYA)

    def test_the_option_is_the_visarga_the_ca_brings_back(self):

        outs = run("vṛkṣas karoti").outcomes
        self.assertEqual(len(outs), 2)
        declined = [s for o in outs for s in o.steps if s.declined]
        self.assertEqual([s.sutra for s in declined], ["8.3.37"])

    def test_kashika_8_3_35_sasah_ksuram_purusah_ksuram(self):
        """Kāśikā on 8.3.35: शशः क्षुरम्। पुरुषः क्षुरम्। अद्भिः प्सातम्। वासः क्षौमम्।

        for text, form in (("śaśas kṣuram", "śaśaḥkṣuram"),
                           ("puruṣas kṣuram", "puruṣaḥkṣuram"),
                           ("adbhis psātam", "adbhiḥpsātam"),
                           ("vāsas kṣaumam", "vāsaḥkṣaumam"),
                           ("puruṣas tsaruḥ", "puruṣaḥtsaruḥ")):
            self.assertForms(text, {form})
            self.assertEqual(len(run(text).outcomes), 1, text)
            self.assertIn("8.3.35", steps(text), text)

            self.assertEqual(step.detail.kind, "pratiṣedha")

    def test_kashika_8_3_35_wins_over_8_3_37_though_later(self):
        """Kāśikā on 8.3.37: पूर्वत्रासिद्धे नास्ति विप्रतिषेधोऽभावादुत्तरस्येति शर्परे
        विसर्जनीयः इत्येतदेव भवति — वासः क्षौमम्, अद्भिः प्सातम्: क and प are

        out = first("vāsas kṣaumam")

        self.assertIn("8.3.34", displaced(first("puruṣas tsaruḥ"), "8.3.35"))
        stripped = tuple(dataclasses.replace(r, overrides=())
                         if r.sutra == "8.3.35" else r for r in RULES)

    def test_a_khar_not_followed_by_a_sar_is_not_8_3_35s(self):

    def test_kashika_8_3_36_vrksah_sete_vrksassete(self):

    def test_kashika_vartika_kharpare_sari_va_visargalopah(self):

        स्थातारः, वृक्षास्स्थातारः — Kaumudī: पक्षे विसर्गे सत्वे च त्रैरूप्यम् —

                refusal = r
        assert refusal
        mid = derive(state, tuple(x for x in RULES if x.sutra in ("8.3.15",)))[0]
        view = lambda rule: View(mid.final, rule.sutra, rule.consumes)
        base = next(r for r in RULES if r.sutra == "8.3.38" and not r.varttika)
        a = list(refusal.find(view(refusal)))
        b = list(base.find(view(base)))

        self.assertEqual(len(b), 1)
        self.assertEqual(a[0].site, b[0].site)
        self.assertEqual(a[0].edits, ())

    def test_every_word_of_the_kaskadi_gana_is_reproduced(self):
        """The gaṇapāṭha on disk lists the kaskādi words as finished words
        (Kāśikā: अविहितलक्षण उपचारः कस्कादिषु द्रष्टव्यः): each one that has a visarga
        before a कु/पु sound is split there, given as two words, and the
        engine must give back the word as the gaṇa spells it — with its स् or
        ष् — through 8.3.48 (or 8.3.45, which agrees). This is the gaṇa read
        against the rule, not the rule against itself."""
        items = [item for g in corpus.ganas_for("8.3.48") for item in g.items]
        self.assertGreater(len(items), 15)
        kupu = set(VARGA["ku"]) | set(VARGA["pu"])
        checked = 0
        for item in items:
            for i, ch in enumerate(item):
                if ch not in ("s", "ṣ") or i == 0 or i + 1 >= len(item):
                    continue
                rest = item[i + 1:]
                if not (rest[0] in ("k", "p")):
                    continue
                left = item[:i] + "s"
                right = rest
                if item[:i].endswith(("ṃ", "n")):
                    continue                      # kāṃskān: 8.3.12, below
                got = forms(f"{left} {right}{{isus}}"
                            if False else f"{left} {right}")
                self.assertIn(item.replace("ḥ", "ḥ"), got, (item, left, right))
                checked += 1
                break
        self.assertGreater(checked, 12)

    def test_kaskadi_partial_a_word_the_list_does_not_spell_is_not_seen(self):
        """PARTIAL (COVERAGE): a pair the gaṇa does not spell is left to 8.3.37."""
        self.assertNotIn("8.3.48", taken(first("kas karoti")))
        self.assertEqual(len(forms("kas karoti")), 2)

class VedicRules(_Base):
    """The rules that are Vedic (`vedic=True`) run only with `veda=True`."""

    def vedic(self, text, rules=RULES, **kw):
        return forms(text, rules, veda=True, **kw)

    def test_they_do_not_run_in_the_language(self):
        """A Vedic rule does not act on ordinary speech — the same words without
        `veda` give only what the common rules give."""
        self.assertEqual(forms("vācas{sasthi} patim"), {"vācaḫpatim",
                                                        "vācaḥpatim"})
        self.assertEqual(self.vedic("vācas{sasthi} patim"), {"vācaspatim"})
        self.assertNotIn("8.3.53", taken(first("vācas{sasthi} patim")))

    def test_the_vedic_rules_are_marked_vedic(self):
        """COVERAGE 'vedic' iff `Rule.vedic`: the two agree."""
        by = {}
        for r in fam.RULES:
            by.setdefault(r.sutra, []).append(r)
        for sutra, status, _ in fam.COVERAGE:
            if status == "vedic":
                self.assertTrue(any(r.vedic for r in by[sutra]), sutra)
        for r in fam.RULES:
            if r.vedic:
                self.assertEqual(
                    dict((s, st) for s, st, _ in fam.COVERAGE)[r.sutra],
                    "vedic", r.sutra)

    def test_kashika_8_3_49_ayah_patram_ayas_patram(self):
        """Kāśikā on 8.3.49: अयःपात्रम्, अयस्पात्रम्। विश्वतः पात्रम्, विश्वतस्पात्रम्।
        उरु णः कारः, उरु णस्कारः — the स्, and the visarga."""
        for text, s_form, v_form in (
                ("ayas pātram", "ayaspātram", "ayaḥpātram"),
                ("viśvatas pātram", "viśvataspātram", "viśvataḥpātram"),
                ("uru ṇas kāraḥ", "uruṇaskāraḥ", "uruṇaḥkāraḥ")):
            got = self.vedic(text)
            self.assertIn(s_form, got, text)
            self.assertIn(v_form, got, text)
            self.assertNotIn(s_form, forms(text), text)   # not without veda

        """Kāśikā on 8.2.77: आस्तीर्णम्। विस्तीर्णम्। विशीर्णम्। अवगूर्णम्। दीव्यति। सीव्यति
        — before a consonant, in the middle of a pada (the न of the ending is
        8.4.2's business, the natva family's)."""
        for text, form in (("div{dhatu}~yati", "dīvyati"),
                           ("sīv{dhatu}~yati".replace("sīv", "siv"), "sīvyati"),
                           ("āstir{dhatu}~na", "āstīrna"),
                           ("viśir{dhatu}~na", "viśīrna"),
                           ("avagur{dhatu}~na", "avagūrna")):
            self.assertForms(text, {form})
            self.assertEqual(steps(text), ["8.2.77"], text)

    def test_kashika_8_2_77_dhator_iti_and_ika_iti_eva(self):
        """Kāśikā: धातोरित्येव — दिवमिच्छति दिव्यति (denominative: no flag);
        इक इत्येव — स्मर्यते, भव्यम्: no इक् before the र्/व्."""
        self.assertNotIn("8.2.77", taken(first("div~yati")))
        for text in ("smar{dhatu}~yate", "bhav{dhatu}~yam"):
            self.assertNotIn("8.2.77", taken(first(text)), text)
            self.assertEqual(first(text).surface, text.replace("{dhatu}", "")
                             .replace("~", ""), text)

    def test_the_penult_is_not_lengthened_twice(self):
        """A penult that arrives long is left alone (no change to make)."""
        self.assertNotIn("8.2.76", taken(first("gīr{dhatu}")))
        self.assertNotIn("8.2.77", taken(first("dīv{dhatu}~yati")))

    def test_kashika_8_2_79_dhuryah_divyam_kuryat_churyat(self):
        """Kāśikā on 8.2.79: धुरं वहति धुर्यः। धुरि साधुर्धुर्यः। दिव्यम्। कुर्यात्।
        छुर्यात् — a भ (flag `bha`), or कुर्, छुर्: no lengthening (8.2.77 is
        refused); and *reph-vakārābhyāṃ bha-viśeṣaṇaṃ kim? pratidīvnā*."""
        bha, kur, chur = BHA_KUR_CHUR
        for text, form in (("dhur{dhatu,bha}~ya", "dhurya"),
                           ("div{dhatu,bha}~ya", "divya"),
                           (f"{kur}{{dhatu}}~yāt", "kuryāt"),
                           (f"{chur}{{dhatu}}~yāt", "churyāt")):
            self.assertForms(text, {form}, pause=False)
            out = first(text, pause=False)
            self.assertNotIn("8.2.77", taken(out), text)
            self.assertIn("8.2.79", taken(out), text)
            step = step_of(out, "8.2.79")
            self.assertEqual(step.detail.kind, "pratiṣedha")
            self.assertIn("8.2.77", [a.sutra for a in step.against])
        # without the flag, the very same letters are lengthened
        self.assertForms("dhur{dhatu}~ya", {"dhūrya"})
        self.assertForms("pratidiv{dhatu}~nā", {"pratidīvnā"})

    def test_8_2_79_refuses_8_2_76_too(self):
        """*Kur* and *chur* at the end of a pada: no *kūḥ*."""
        out = first("kur{dhatu}")
        self.assertNotIn("8.2.76", taken(out))
        self.assertEqual(out.surface, "kuḥ")
        self.assertEqual(first("pur{dhatu}").surface, "pūḥ")

    def test_8_2_78_is_not_a_rule(self):
        """8.2.78 is `scope` (the letters are inside a dhātu); nothing claims it."""
        self.assertNotIn("8.2.78", [r.sutra for r in fam.RULES])
        self.assertEqual(dict((s, st) for s, st, _ in fam.COVERAGE)["8.2.78"],
                         "scope")

        for sutra, work, phrase in IN_THE_COMMENTARY:
            # the printed text has a doubled space here and there (रक्तं  वासः)
            # and carries Vedic accent marks (अग॑न्म॒) the quotation leaves off
            def plain(t):
                return " ".join(t.replace("\u0951", "").replace("\u0952", "").split())
            text = plain(corpus.commentary_on(sutra, work) or "")
            self.assertIn(plain(phrase), text, (sutra, work, phrase))

    def test_every_quotation_the_module_makes_is_verbatim(self):
        """`_q` and `_bare` register each quotation; every one is read back from
        the corpus (a reason that the tradition did not give is not given)."""
        self.assertGreater(len(fam.QUOTED), 30)
        for sutra, work, quote in fam.QUOTED:
            if work == "varttika":
                found = any(quote in v.text for v in corpus.varttikas_on(sutra))
            else:
                found = quote in (corpus.commentary_on(sutra, work) or "")
            self.assertTrue(found, (sutra, work, quote))

    def test_the_vartikas_are_the_corpus_own_or_a_quoted_line(self):
        """Every rule with `authority=VARTTIKA` carries a `varttika` that is a
        vārttika of the corpus, or the Kaumudī's marked line."""
        for r in fam.RULES:
            if r.authority != "vārttika":
                continue
            corpus_texts = {v.text.strip(" ।")
                            for v in corpus.varttikas_on(r.sutra)}
            quoted = {q for s, w, q in fam.QUOTED if s == r.sutra}
            self.assertTrue(r.varttika in corpus_texts or r.varttika in quoted,
                            (r.sutra, r.varttika))
            self.assertTrue(r.name == r.varttika)

    def test_the_names_of_the_rules_are_the_sutras_own_words(self):
        """A rule's name is the sūtra's text in Devanāgarī from the corpus — none
        is typed."""
        for r in fam.RULES:
            if r.authority == "vārttika":
                continue
            self.assertEqual(r.name, trace.deva(
                corpus.load_vidyut_sutrapatha()[r.sutra].text), r.sutra)

    def test_the_gana_lists_are_the_commentarys_words(self):
        """The word lists this module asks of `visarjaniya` and `ru_adesa` are
        the Kāśikā's: read back from the corpus, not just trusted."""
        pairs = (
            ("8.3.43", "द्विस् त्रिस् चतुर्", DVIS_THREE),
            ("8.3.53", "पति पुत्र पृष्ठ पार पद पयस् पोष", PATI_SEVEN),
            ("8.2.70", "अम्नस् ऊधस् अवस्", UBHAYATHA_THREE),
        )
        for sutra, line, expected in pairs:
            self.assertIn(line, corpus.commentary_on(sutra, "kashika"), sutra)
            self.assertEqual(tuple(to_iast(w) for w in line.split()),
                             tuple(expected), sutra)

#: The scope, as the sūtras it covers — 62 sūtras.
SCOPE = ([f"8.2.{n}" for n in range(62, 82)] + [f"8.3.{n}" for n in range(8, 23)]
         + [f"8.3.{n}" for n in range(34, 55)]
         + ["6.1.113", "6.1.114", "6.3.111"]
         + [f"6.1.{n}" for n in range(132, 136)])


class Declarations(unittest.TestCase):

    def test_the_rulebook_is_sound(self):
        self.assertEqual(rulebook.problems(), [])

    def test_coverage_is_exactly_the_scope(self):
        ids = [s for s, _, _ in fam.COVERAGE]
        self.assertEqual(sorted(ids), sorted(SCOPE))
        self.assertEqual(len(ids), len(set(ids)))

    def test_every_status_is_honest(self):
        """A rule is 'rule', 'partial' or 'vedic'; a sūtra with no rule is
        'scope' and says why; 'vedic' means the rule is `vedic`."""
        have = {}
        for r in fam.RULES:
            have.setdefault(r.sutra, []).append(r)
        for sutra, status, note in fam.COVERAGE:
            self.assertIn(status, rulebook.COVERAGE_STATUS)
            if status == "scope":
                self.assertNotIn(sutra, have, sutra)
                self.assertGreater(len(note), 40, sutra)
            else:
                self.assertIn(sutra, have, sutra)
            if status == "partial":
                self.assertGreater(len(note), 40, sutra)
            if status == "vedic":
                self.assertTrue(any(r.vedic for r in have[sutra]), sutra)
            if status in ("rule", "partial"):
                self.assertTrue(any(not r.vedic for r in have[sutra]), sutra)
        for r in fam.RULES:
            self.assertIn(r.sutra, [s for s, _, _ in fam.COVERAGE])

    def test_every_rule_carries_the_family_tags(self):
        """README: the ru/visarga rules 'visarga'; the consonant rules 'hal'."""
        for r in fam.RULES:
            self.assertTrue({"visarga", "hal"} & set(r.families), r.sutra)

    def test_every_override_is_a_sutra_or_a_tag_and_says_why_in_quotation(self):
        known = corpus.load_vidyut_sutrapatha()
        tags = set()
        for group in rulebook.by_family().values():
            for r in group:
                tags.update(r.families)
        registered = {q for _, _, q in fam.QUOTED}
        for r in fam.RULES:
            for target, why in r.overrides:
                if target.startswith("@"):
                    self.assertIn(target[1:], tags, (r.sutra, target))
                else:
                    self.assertIn(target, known, (r.sutra, target))
                self.assertTrue(any(q in why for q in registered),
                                (r.sutra, target))

    def test_a_sutra_tag_belongs_to_one_rule_only(self):
        """`sutra:8.3.36` marks the sūtra itself, never its vārttika — so that
        naming one does not name the other."""
        seen = {}
        for r in fam.RULES:
            for tag in r.families:
                if tag.startswith("sutra:"):
                    self.assertNotIn(tag, seen, tag)
                    seen[tag] = r
        for tag, r in seen.items():
            self.assertEqual("sutra:" + r.sutra, tag)
            self.assertEqual(r.varttika, "")

    def test_two_rules_never_share_a_sutra_and_a_varttika(self):
        keys = [(r.sutra, r.varttika) for r in fam.RULES]
        self.assertEqual(len(keys), len(set(keys)))

    def test_the_vedic_rules_are_only_run_in_the_veda(self):
        """`Rule.vedic` is what the engine reads: with `veda=False` none of
        them is even asked."""
        from src.astadhyayi.sandhi.engine import collect
        state = parse("vācas{sasthi} patim")
        self.assertFalse(state.veda)
        offered = {r.sutra for r, _ in collect(state, RULES)}
        self.assertNotIn("8.3.53", offered)
        vedic = dataclasses.replace(state, veda=True)
        after = derive(vedic, RULES)[0].final
        self.assertEqual(after.joined(), "vācaspatim")

    def test_the_flags_are_documented(self):
        """Every flag the module reads by name is in its docstring's list."""
        import re
        with open(fam.__file__, encoding="utf-8") as handle:
            source = handle.read()
        named = set(re.findall(r'_flag\(v, [a-z_.]+, "([a-z_]+)"', source))
        named |= set(re.findall(r'_asked\(v, "([a-z_]+)"', source))
        doc = fam.__doc__
        for flag in sorted(named):
            self.assertIn(flag, doc, flag)


class Agreement(_Base):
    """The project already codifies these sūtras as questions

    def test_the_visarga_table_and_the_engine_choose_the_same_sutra(self):

class Termination(unittest.TestCase):
    """Whatever the flags and whatever the sounds, the family's rules end, and
    end with sounds — and cite sūtras that exist (the engine's own test does
    this for the bare letters; here for the flags this family adds)."""

    FINALS = ("s", "r", "n", "m", "ṣ", "h", "y", "v", "o", "a", "ā", "i", "u")
    INITIALS = ("a", "ā", "i", "u", "e", "k", "kh", "p", "c", "t", "ś", "s",
                "r", "d", "v", "h", "ḍh")
    FLAGSETS = ("", "{dhatu}", "{gati}", "{apratyaya}", "{isus,samartha}",
                "{sasthi}", "{pancami}", "{stem:etad}", "{krtvo_artha}",
                "{avyaya}", "{sup}", "{sakatayana}", "{padapuranam}",
                "{kvasu}", "{bha,dhatu}")

    def test_nothing_loops_or_stops_short(self):
        known = corpus.load_vidyut_sutrapatha()
        checked = 0
        for veda in (False, True):
            for final in self.FINALS:
                for flags in self.FLAGSETS:
                    for initial in self.INITIALS[::2 if not veda else 3]:
                        first_word = f"kata{final}{flags}"
                        second = f"{initial}ta"
                        result = sandhi(f"{first_word} {second}", rules=RULES_YAN,
                                        veda=veda)
                        self.assertTrue(result.outcomes)
                        for o in result.outcomes:
                            self.assertNotIn("cycling", o.stopped, (first_word,
                                                                    second))
                            self.assertNotIn("cap", o.stopped, (first_word,
                                                                second))
                            for step in o.steps:
                                for s in step.sutras:
                                    self.assertIn(s, known)
                            trace.outcome_dict(o)             # renders
                        checked += 1
        self.assertGreater(checked, 1000)

    def test_a_call_stays_in_the_millisecond_range(self):
        """The engine is asked at every step of every rule, so a family that
        added a slow `find` would show here. Generous bound: a ~ms call."""
        import time
        sandhi("rāmas ca")
        began = time.perf_counter()
        for _ in range(20):
            sandhi("rāmas ca")
            sandhi("devās iha")
            sandhi("vṛkṣas karoti")
        per_call = (time.perf_counter() - began) / 60
        self.assertLess(per_call, 0.08, f"{per_call * 1000:.1f} ms per call")

    def test_options_multiply_only_where_the_grammar_offers_them(self):
        """A whole clause of junctions returns every course and is capped, not
        stuck."""
        text = "devās iha haris śete vṛkṣas karoti tiras{gati} kartā"
        result = sandhi(text, rules=RULES)
        self.assertTrue(1 < len(result.outcomes) <= 64)
        for o in result.outcomes:
            self.assertNotIn("cycling", o.stopped)
